#!/usr/bin/env bash
set -euo pipefail

SOURCE_DIR="/var/www"
BACKUP_DIR="/var/backups/myapp"
PREFIX="myapp"
KEEP=7
LOG_TAG="backup-dir"

log() { logger -t "$LOG_TAG" -p local0.info "$*"; }

mkdir -p "$BACKUP_DIR"
timestamp="$(date +%Y%m%d-%H%M%S)"
archive="$BACKUP_DIR/${PREFIX}-${timestamp}.tar.gz"

log "Starting backup of $SOURCE_DIR to $archive"

# maybe lock
exec 9>"$BACKUP_DIR/.lock"
if ! flock -n 9; then log "Another backup is running; exiting."; exit 0; fi

if [[ ! -d "$SOURCE_DIR" ]]; then log "ERROR: source directory $SOURCE_DIR does not exist"; exit 1; fi

tar -czf "$archive" -C "$(dirname "$SOURCE_DIR")" "$(basename "$SOURCE_DIR")"
log "Backup completed: $archive"

# retention: list archives matching prefix sorted newest first, skip first KEEP, delete rest
mapfile -t old_backups < <(ls -1t "$BACKUP_DIR/${PREFIX}-"*.tar.gz 2>/dev/null | tail -n +$((KEEP + 1)))
for old in "${old_backups[@]}"; do
  log "Removing old backup: $old"
  rm -f -- "$old"
done

log "Retention complete; kept $KEEP most recent backups"
