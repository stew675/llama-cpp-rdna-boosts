#!/usr/bin/env bash
set -euo pipefail

SOURCE_DIR="/path/to/backup"
BACKUP_DIR="/var/backups/myapp"
PREFIX="myapp"
KEEP=7
LOG_TAG="backup-dir"

mkdir -p "$BACKUP_DIR"
timestamp="$(date +%Y%m%d-%H%M%S)"
archive="$BACKUP_DIR/${PREFIX}-${timestamp}.tar.gz"

logger -t "$LOG_TAG" "Starting backup of $SOURCE_DIR to $archive"

if tar -czf "$archive" -C "$(dirname "$SOURCE_DIR")" "$(basename "$SOURCE_DIR")"; then
  logger -t "$LOG_TAG" "Backup completed: $archive ($(du -h "$archive" | cut -f1))"
else
  rc=$?
  logger -t "$LOG_TAG" "Backup failed for $SOURCE_DIR (exit $rc)"
  exit "$rc"
fi

# retention: keep newest KEEP archives matching prefix
mapfile -t old_backups < <(ls -1t "$BACKUP_DIR/${PREFIX}-"*.tar.gz 2>/dev/null | tail -n +$((KEEP + 1)))
for old in "${old_backups[@]}"; do
  if rm -f -- "$old"; then
    logger -t "$LOG_TAG" "Removed old backup: $old"
  else
    logger -t "$LOG_TAG" "Failed to remove old backup: $old"
  fi
done
